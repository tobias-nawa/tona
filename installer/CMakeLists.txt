# The code of `tona setup` (the program is built by engine/CMakeLists.txt) and its tests: a project of its own, so the
# tests need a C++ compiler, CMake, Python 3 and the OpenSSL development files and no HIP. Build and test:
#   cmake -S installer -B work/build/installer -G Ninja && cmake --build work/build/installer && ctest --test-dir work/build/installer
cmake_minimum_required(VERSION 3.28)
project(tona_install LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release)
endif()

# Hardening of the release program (shared with the engine; the release build sets the option)
option(TONA_HARDEN "Harden the host code: FORTIFY_SOURCE=3, stack protector, CET, PIE, full RELRO" OFF)
if(TONA_HARDEN)
    include(${CMAKE_CURRENT_SOURCE_DIR}/../engine/cmake/hardening.cmake)
endif()

# The updater of tona (engine/update/), for the first installation of a release: the same signature check, download
# and layout, no second path. PUBLIC httplib macro as in engine/CMakeLists.txt.
find_package(OpenSSL REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
include(${CMAKE_CURRENT_SOURCE_DIR}/../engine/update/sources.cmake)
add_library(tona_update STATIC ${TONA_UPDATE_SOURCES})
target_include_directories(tona_update PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/../engine)
target_compile_definitions(tona_update PUBLIC CPPHTTPLIB_OPENSSL_SUPPORT)
target_link_libraries(tona_update PUBLIC OpenSSL::SSL OpenSSL::Crypto pthread)
if(APPLE)  # cpp-httplib reads the system certificates through these (as in engine/CMakeLists.txt)
    target_link_libraries(tona_update PUBLIC "-framework CoreFoundation" "-framework Security")
endif()

# The offer data (tasks, recommendations, model cards) of engine/offer/, shared with the client
include(${CMAKE_CURRENT_SOURCE_DIR}/../engine/offer/offer.cmake)
include(${CMAKE_CURRENT_SOURCE_DIR}/core.cmake)  # the core library, shared with the engine project (the client)
include(${CMAKE_CURRENT_SOURCE_DIR}/setup.cmake)  # the guided setup and FTXUI, shared with the engine project
