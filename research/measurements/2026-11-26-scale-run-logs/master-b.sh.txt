#!/usr/bin/env bash
# Scale run 2026-11-26: blocks b, c1, c3, c4, e, f, f2, i (no g). First scale run on the Ryzen 9 9950X; compared with 2026-11-14 (9600X).
# Differs from the 11-14 drivers only in paths, build name and OMP_NUM_THREADS (no longer fixed to 6: the server sets the 16 physical cores). Start as its own background job.
# Every GPU step goes through the GPU line (run.sh, the golden scripts and baseline.sh take their own turn).
S=/var/tmp/claude/scale1126; R="$S/run.sh"; W=<worktree>
cd "$W" || exit 1
# shellcheck disable=SC1091
source toolchain/env.sh; export TONA_DIAG=1
B=work/build/scale1126
vram() { echo "VRAM-desktop $1 $(date +%T) $(( $(cat /sys/class/drm/card1/device/mem_info_vram_used) / 1048576 )) MiB"; }
step() { echo "STEP $1 $(date +%T)"; vram "$1"; }
# test conditions of a block, before and after, into the block's own log (sums only, no program names)
tc() { { echo "## $2 $1 $(date +%F' '%T)"; python3 tools/hw_report.py; python3 tools/quiet_check.py; echo "quiet_check exit $?"; } >> "$S/$1.tc.log" 2>&1; }
gold() { local s rc; s=$(date +%s); echo "=== $1 $(date +%T)"; shift; "$@"; rc=$?; echo "=== rc=$rc $(( $(date +%s)-s )) s"; return $rc; }
cv() {
  local l=$1; shift; sleep 60; step "$l"; tc "$l" before
  RUN_LABEL=$l "$R" python3 -I "$S/curve.py" "$l" scale1126 "$@" > "$S/$l.out" 2>&1; tc "$l" after
  grep -h "expert slots" "$S/$l"-server-*.log | sed "s/^/SLOTS $l: /"
  grep -h "huge pages" "$S/$l"-server-*.log | sed "s/^/HUGE $l: /"
}
abb() { # label, extra ab.py args...
  local l=$1; shift; sleep 60; step "$l"; tc "$l" before
  RUN_LABEL=$l "$R" python3 scripts/ab.py --a scale1126 --b scale1126 --load-log on --out "work/ab/n2-$l" "$@" > "$S/$l.out" 2>&1; tc "$l" after
  grep -h "expert slots" work/ab/n2-"$l"-logs/* 2>/dev/null | sed "s/^/SLOTS $l: /" | sort | uniq -c
}
M=<repo>/work/models
cv b mid,long --env TONA_DETERMINISTIC=1 --env TONA_DIAG=1
cv c1 set --env TONA_DIAG=1
cv c3 mid,long --env TONA_DIAG=1
abb o-coder --workload server --targets 2000:200,16000:100,64000:50 --ctx 262144 --min-rounds 2 --max-rounds 2 --fixed-rounds \
  --why-size "the offer.json rows of the Coder (2K, 16K, 64K code summary, MTP); same targets as 2026-11-20-coder-decode.md; A/A gives the spread; 2 rounds is the minimum" \
  --expect decode:-3..+3% --expect prefill:-3..+3% --expect-source "A/A, same build: B against A is 0 within the known noise"
abb o-ornith --model $M/ornith-1.5-35b-a3b/Ornith-1.5-35B-Q4_K_M.gguf --drafter none --workload server --targets 2000:200,16000:200,64000:100 --ctx 262144 \
  --min-rounds 2 --max-rounds 3 \
  --why-size "the offer.json rows of Ornith (2K, 16K, 64K code summary, MTP head from the file); targets as 2026-11-20-ornith-bolt-clean-state.md; 2 to 3 rounds from noise.toml" \
  --expect decode:-3..+3% --expect prefill:-3..+3% --expect-source "A/A, same build: B against A is 0 within the known noise"
abb o-bolt --model $M/nemotron-bolt-74b-a9b-24gb/NVIDIA-Nemotron-3.5-Bolt-74B-A9B-24GB.gguf --drafter none --workload server --targets 2000:200,16000:200,64000:100 --ctx 262144 \
  --min-rounds 2 --max-rounds 2 \
  --why-size "the offer.json rows of Bolt (2K, 16K, 64K code summary, no drafter, history drafts); targets as 2026-11-20-ornith-bolt-clean-state.md; 2 rounds is the minimum of ab.py" \
  --expect decode:-3..+3% --expect prefill:-3..+3% --expect-source "A/A, same build: B against A is 0 within the known noise"
cv c4 huge --env TONA_DIAG=1
# e: R8 at ctx 262144 (the service default, DRIVE_CTX default in drive.py): the overflow of the pinned experts is registered there
sleep 60; step e; tc e before; RUN_LABEL=e "$R" python3 -I "$S/drive.py" press 10 r8 scale1126 DRIVE_ULIMIT_C=unlimited TONA_DIAG=1 > "$S/r8.out" 2>&1; tc e after
grep -h "expert slots" "$S/r8.server.log" | sed "s/^/SLOTS e: /"; grep -h "huge pages" "$S/r8.server.log" | sed "s/^/HUGE e: /"
cv f server --targets 250000:100 --ctx 262144 --env TONA_DIAG=1
cv f2 server --targets 1000000:200 --ctx 1048576 --env TONA_SEQS=1 --env TONA_DIAG=1
sleep 60; step i; tc i before
bash "$S/i.sh" > "$S/i.out" 2>&1
tc i after
step finished
