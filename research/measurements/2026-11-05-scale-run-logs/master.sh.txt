#!/usr/bin/env bash
S=/var/tmp/claude/scale1105; R=$S/run.sh; W=<worktree>
cd $W; source toolchain/env.sh; export TONA_DIAG=1
B=work/build/merged
step() { echo "STEP $1 $(date +%T)"; }
gold() { local s=$(date +%s); echo "=== $1 $(date +%T)"; shift; "$@"; local rc=$?; echo "=== rc=$rc $(( $(date +%s)-s )) s"; return $rc; }
step a
{
gold "prefill dflash" env GOLDEN_MODE=prefill GOLDEN_DRAFTER=dflash scripts/golden.sh compare $B &&
gold "gpu" scripts/golden.sh compare $B &&
gold "cpu plan" env GOLDEN_PLAN=cpu scripts/golden.sh compare $B &&
gold "server" scripts/golden-server.sh compare $B &&
gold "prefill NB=1" env GOLDEN_MODE=prefill GOLDEN_PROMPT_NB=1 scripts/golden.sh compare $B && echo ALLOK
} > $S/a.log 2>&1
grep -q ALLOK $S/a.log || { echo "golden failed"; exit 1; }
cv() { sleep 60; step $1; local l=$1; shift; $R python3 -I $S/curve.py $l merged "$@" > $S/$l.out 2>&1; }
cv b mid,long --env TONA_DETERMINISTIC=1 --env TONA_DIAG=1
cv c1 set --env TONA_DIAG=1
cv c3 mid,long --env TONA_DIAG=1
cv c4 huge --env TONA_DIAG=1
sleep 60; step e; $R python3 -I $S/drive.py press 10 r8 merged DRIVE_ULIMIT_C=unlimited TONA_DIAG=1 > $S/r8.out 2>&1
cv f server --targets 250000:100 --ctx 262144 --env TONA_DIAG=1
rm -f $S/route1m.trace
cv f2 server --targets 1000000:200 --ctx 1048576 --env TONA_SEQS=1 --env TONA_DIAG=1 --env TONA_ROUTE_TRACE=$S/route1m.trace
nd() { sleep 60; step $1; local l=$1 t=$2 len=$3 ctx=$4; shift 4; NEEDLE_TEXTS=$t $R python3 -I $S/needle.py $l $len $ctx TONA_DIAG=1 "$@" > $S/$l.out 2>&1; }
for spec in "0 on off" "1 off on" "2 on off"; do
  set -- $spec; t=$1; shift
  for arm in "$@"; do
    if [ $arm = off ]; then nd n128-t$t-off $t 131072 140000 TONA_PREFILL_FP8=0; else nd n128-t$t-on $t 131072 140000; fi
  done
done
nd n1m-t0-on 0 1000000 1048576 TONA_SEQS=1 TONA_PROMPT_RAM_GB=0
nd n1m-t0-off 0 1000000 1048576 TONA_SEQS=1 TONA_PROMPT_RAM_GB=0 TONA_PREFILL_FP8=0
cv h3 server --targets 1000000:200 --ctx 1048576 --env TONA_SEQS=1 --env TONA_DIAG=1 --env TONA_PREFILL_FP8=0
step done
